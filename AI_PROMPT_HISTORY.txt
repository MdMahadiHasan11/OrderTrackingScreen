Here is a clean, runnable Next.js page component based on your request. It features a centered mobile container, a top header with a back button, order title with order number, a help button, and a 3-tab switch component to toggle between the delayed, missing, and unavailable states.

TypeScript
'use client'

import { useState } from 'react'
import { ArrowLeft, CircleHelp } from 'lucide-react'

const states = {
  delayed: {
    eyebrow: 'Delivery update',
    title: 'Your order is delayed',
  },
  missing: {
    eyebrow: 'Delivered today',
    title: 'We couldn’t find your package?',
  },
  unavailable: {
    eyebrow: 'Order confirmed',
    title: 'Tracking isn’t available yet',
  },
} as const

type StateKey = keyof typeof states

export default function Page() {
  const [state, setState] = useState<StateKey>('delayed')
  const current = states[state]

  return (
    <main className="min-h-screen bg-[#f6f8f9] px-4 pt-6 font-sans text-[#17212b]">
      {/* Centered mobile container */}
      <div className="mx-auto max-w-[430px]">
        
        {/* Top Header */}
        <header className="mb-[18px] flex items-center justify-between">
          <button 
            className="grid size-10 cursor-pointer place-items-center rounded-[10px] bg-transparent p-2 text-[#17212b] hover:bg-[#eaf0f1]"
            aria-label="Go back"
          >
            <ArrowLeft className="size-5" />
          </button>
          
          <div className="flex flex-col text-center text-[15px] font-bold">
            <span>Order tracking</span>
            <small className="text-[11px] font-medium text-[#6d7b88]">Order #18492</small>
          </div>
          
          <button 
            className="grid size-10 cursor-pointer place-items-center rounded-[10px] bg-transparent p-2 text-[#17212b] hover:bg-[#eaf0f1]"
            aria-label="Get help"
          >
            <CircleHelp className="size-5" />
          </button>
        </header>

        {/* 3-Tab Switch Component */}
        <div className="mb-3.5 grid grid-cols-3 gap-0.5 rounded-xl bg-[#e9eff0] p-[3px]" role="tablist">
          {(Object.keys(states) as StateKey[]).map((key) => (
            <button
              key={key}
              role="tab"
              aria-selected={state === key}
              className={
                state === key
                  ? 'cursor-pointer rounded-[9px] bg-white py-[9px] text-[11px] font-bold text-[#17212b] shadow-[0_2px_7px_#1b323312]'
                  : 'cursor-pointer rounded-[9px] bg-transparent py-[9px] text-[11px] font-bold text-[#6d7b88]'
              }
              onClick={() => setState(key)}
            >
              {key === 'delayed' ? 'Delayed' : key === 'missing' ? 'Not received' : 'No tracking'}
            </button>
          ))}
        </div>

        {/* Dynamic Display based on Selected State */}
        <div className="rounded-[15px] border border-[#e8edf0] bg-white p-4 text-center">
          <p className="text-[10px] font-extrabold uppercase tracking-[0.1em] text-[#6d7b88]">{current.eyebrow}</p>
          <h1 className="mt-1 text-base font-bold">{current.title}</h1>
        </div>

      </div>
    </main>
  )
}

  AlertCircle,
  ArrowLeft,
  CircleHelp,
  Clock3,
  PackageCheck,

    time: 'at 2:14 PM',

    eyebrow: 'Order confirmed',
    title: 'Tracking isn’t available yet',
    detail: 'We’ll add updates here as soon as the carrier scans your package.',

    tone: 'blue',
    badge: 'On schedule',

type StateKey = keyof typeof states

                    </strong>

                    </small>
                  </span>

              )

          </ol>
